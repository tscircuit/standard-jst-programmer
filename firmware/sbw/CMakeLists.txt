cmake_minimum_required(VERSION 3.13)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(standard_jst_sbw C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT DEFINED SBW_USB_VID OR NOT DEFINED SBW_USB_PID)
  message(FATAL_ERROR "Set SBW_USB_VID and SBW_USB_PID to a USB identity you are authorized to use; CI IDs are test-only")
endif()
add_executable(standard-jst-sbw main.c protocol.c session.c usb_descriptors.c vendor/sbw_device.c vendor/sbw_jtag.c vendor/sbw_transport.c)
target_include_directories(standard-jst-sbw PRIVATE . vendor)
target_compile_definitions(standard-jst-sbw PRIVATE SBW_USB_VID=${SBW_USB_VID} SBW_USB_PID=${SBW_USB_PID})
target_link_libraries(standard-jst-sbw pico_stdlib pico_unique_id hardware_sync tinyusb_device tinyusb_board)
pico_enable_stdio_usb(standard-jst-sbw 0)
pico_enable_stdio_uart(standard-jst-sbw 0)
pico_add_extra_outputs(standard-jst-sbw)
